EMPLOYEE DATA EXECUTION QUERIES:

SELECT 
    CURRENT_USER() AS USER_NAME,
    CURRENT_ROLE() AS ACTIVE_ROLE,
    CURRENT_WAREHOUSE() AS ACTIVE_WAREHOUSE,
    CURRENT_DATABASE() AS ACTIVE_DATABASE,
    CURRENT_SCHEMA() AS ACTIVE_SCHEMA;

SHOW DATABASES;

CREATE DATABASE EMPLOYEE_DB;

SHOW DATABASES LIKE 'EMPLOYEE_DB';

USE EMPLOYEE_DB;


CREATE SCHEMA IF NOT EXISTS RAW;


USE SCHEMA EMPLOYEE_DB.RAW;

SHOW SCHEMAS IN DATABASE EMPLOYEE_DB;

CREATE WAREHOUSE IF NOT EXISTS EMPLOYEE_WH
    WAREHOUSE_SIZE = 'XSMALL'
    AUTO_SUSPEND = 60
    AUTO_RESUME = TRUE;


USE WAREHOUSE EMPLOYEE_WH;

SHOW WAREHOUSES LIKE 'EMPLOYEE_WH';

CREATE OR REPLACE TABLE EMPLOYEES (
    EMPLOYEE_ID INT PRIMARY KEY,
    EMPLOYEE_NAME VARCHAR(100),
    DEPARTMENT VARCHAR(50),
    JOB_TITLE VARCHAR(100),
    CITY VARCHAR(50),
    JOINING_DATE DATE,
    MONTHLY_SALARY NUMBER(10,2),
    PERFORMANCE_RATING INT,
    EMPLOYMENT_STATUS VARCHAR(30)
);


SHOW TABLES LIKE 'EMPLOYEES' IN SCHEMA EMPLOYEE_DB.RAW;

DESCRIBE TABLE EMPLOYEE_DB.RAW.EMPLOYEES;

CREATE FILE FORMAT IF NOT EXISTS EMPLOYEES_CSV_FORMAT
TYPE = CSV 
SKIP_HEADER = 1
FIELD_OPTIONALLY_ENCLOSED_BY = '"'
EMPTY_FIELD_AS_NULL = TRUE;


SHOW FILE FORMATS IN SCHEMA EMPLOYEE_DB.RAW;


CREATE STAGE IF NOT EXISTS EMPLOYEES_STAGE
FILE_FORMAT = EMPLOYEES_CSV_FORMAT;

SHOW STAGES LIKE 'EMPLOYEES_STAGE' IN SCHEMA EMPLOYEE_DB.RAW;

SHOW FILE FORMATS LIKE 'EMPLOYEES_CSV_FORMAT'
IN SCHEMA EMPLOYEE_DB.RAW;


LIST @EMPLOYEE_DB.RAW.EMPLOYEES_STAGE;


COPY INTO EMPLOYEE_DB.RAW.EMPLOYEES
FROM @EMPLOYEE_DB.RAW.EMPLOYEES_STAGE
FILE_FORMAT = (
    FORMAT_NAME = 'EMPLOYEE_DB.RAW.EMPLOYEES_CSV_FORMAT'
)
ON_ERROR = 'ABORT_STATEMENT';


SELECT *
FROM EMPLOYEE_DB.RAW.EMPLOYEES
ORDER BY EMPLOYEE_ID;

SELECT COUNT(*) AS TOTAL_EMPLOYEES
FROM EMPLOYEE_DB.RAW.EMPLOYEES;

SELECT EMPLOYEE_ID, COUNT(*) AS RECORD_COUNT
FROM EMPLOYEE_DB.RAW.EMPLOYEES
GROUP BY EMPLOYEE_ID
HAVING COUNT(*) > 1;


SELECT SUM(MONTHLY_SALARY) AS TOTAL_MONTHLY_SALARY
FROM EMPLOYEE_DB.RAW.EMPLOYEES;

SELECT
    CITY,
    SUM(MONTHLY_SALARY) AS TOTAL_MONTHLY_SALARY
FROM EMPLOYEE_DB.RAW.EMPLOYEES
GROUP BY CITY
ORDER BY TOTAL_MONTHLY_SALARY DESC;

SELECT
    DEPARTMENT,
    COUNT(*) AS TOTAL_EMPLOYEES,
    AVG(MONTHLY_SALARY) AS AVG_SALARY,
    MAX(MONTHLY_SALARY) AS HIGHEST_SALARY,
    MIN(MONTHLY_SALARY) AS LOWEST_SALARY
FROM EMPLOYEE_DB.RAW.EMPLOYEES
GROUP BY DEPARTMENT
ORDER BY AVG_SALARY DESC;
