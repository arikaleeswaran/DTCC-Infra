SNOWFLAKE SQL QUERIES:

SHOW DATABASES;

CREATE DATABASE SALES_DB;

SHOW DATABASES LIKE 'SALES_DB';

USE SALES_DB;

CREATE SCHEMA IF NOT EXISTS RAW;

USE SCHEMA SALES_DB.RAW;

SHOW SCHEMAS IN DATABASE SALES_DB;

CREATE WAREHOUSE IF NOT EXISTS SALES_WH;

USE WAREHOUSE SALES_WH;

SHOW WAREHOUSES LIKE 'SALES_WH';

ALTER WAREHOUSE SALES_WH SET
  WAREHOUSE_SIZE = 'XSMALL'
  AUTO_SUSPEND = 60
  AUTO_RESUME = TRUE;


SELECT CURRENT_WAREHOUSE();


CREATE TABLE IF NOT EXISTS SALES (
    ORDER_ID       INT PRIMARY KEY,
    ORDER_DATE     DATE,
    CITY           VARCHAR(50),
    CATEGORY       VARCHAR(50),
    PRODUCT        VARCHAR(100),
    QUANTITY       INT,
    UNIT_PRICE     NUMBER(10,2),
    SALES_CHANNEL  VARCHAR(20),
    ORDER_STATUS   VARCHAR(20)
);


SHOW TABLES LIKE 'SALES' IN SCHEMA SALES_DB.RAW;


DESCRIBE TABLE SALES_DB.RAW.SALES;

CREATE FILE FORMAT IF NOT EXISTS SALES_CSV_FORMAT
TYPE = CSV 
SKIP_HEADER = 1
FIELD_OPTIONALLY_ENCLOSED_BY = '"'
EMPTY_FIELD_AS_NULL = TRUE;


SHOW FILE FORMATS IN SCHEMA SALES_DB.RAW;


CREATE STAGE IF NOT EXISTS SALES_STAGE
FILE_FORMAT = SALES_CSV_FORMAT;

SHOW STAGES LIKE 'SALES_STAGE' IN SCHEMA SALES_DB.RAW;

SHOW FILE FORMATS LIKE 'SALES_CSV_FORMAT'
IN SCHEMA SALES_DB.RAW;



LIST @SALES_DB.RAW.SALES_STAGE;

COPY INTO SALES_DB.RAW.SALES
FROM @SALES_DB.RAW.SALES_STAGE
FILE_FORMAT = (
    FORMAT_NAME = 'SALES_DB.RAW.SALES_CSV_FORMAT'
)
ON_ERROR = 'ABORT_STATEMENT';


SELECT *
FROM SALES_DB.RAW.SALES
ORDER BY ORDER_ID;

SELECT COUNT(*)
FROM SALES_DB.RAW.SALES;


SELECT ORDER_ID, COUNT(*) AS RECORD_COUNT
FROM SALES_DB.RAW.SALES
GROUP BY ORDER_ID
HAVING COUNT(*) > 1;


SELECT  SUM(QUANTITY * UNIT_PRICE)  AS TOTAL_REVENUE
FROM SALES_DB.RAW.SALES;

SELECT CITY, SUM(QUANTITY * UNIT_PRICE) AS TOTAL_REVENUE
FROM SALES_DB.RAW.SALES
GROUP BY CITY
ORDER BY TOTAL_REVENUE DESC;

DROP DATABASE IF EXISTS SALES_DB;

DROP WAREHOUSE IF EXISTS SALES_WH;
